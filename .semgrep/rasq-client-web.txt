── rasq-client-web-fetch-gate ──
接続先を確かめてから開き、送ったあとにリダイレクト先を確かめる (wasm_plugin_worker_web.dart の netFetch と同じ形):
function netFetchOk(args) {
  const url = String(args.url || "");
  if (!hostPermitted(hostOf(url))) throw new HostFnError("E01", "host not permitted: " + hostOf(url));
  // ok: rasq-client-web-fetch-gate
  const xhr = new XMLHttpRequest();
  xhr.open("GET", url, false);
  // ok: rasq-client-web-fetch-gate
  try { xhr.send(body); } catch (e) { throw new HostFnError("B01", "fetch failed: " + e); }
  if (xhr.responseURL && !hostPermitted(hostOf(xhr.responseURL))) throw new HostFnError("E01", "redirected");
  return xhr.status;
}
